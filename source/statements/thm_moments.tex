\begin{theorem}[One-use purity and trace bounds]\label{thm:moments}
For forward homogenization, define $A_M=2^M\sigma_{M,1}-I$ and let $\langle X\rangle=2^{-M}\Tr X$. Then $A_M$ is a Hermitian contraction and
\begin{align}
 a_M:=\langle A_M^2\rangle&=1-(1-u^2)^M,\label{eq:purity}\\
 \langle A_M^4\rangle&\le(3M^2+M)u^4.\label{eq:fourth}
\end{align}
Writing $d_1=T(\sigma_{M,1},\tau^{\otimes M})$, one has
\begin{equation}
 \frac{a_M}{2}\le d_1\le\frac{\sqrt{a_M}}2\le\frac{u\sqrt M}{2}.
 \label{eq:basicreturn}
\end{equation}
If $d_1\le1/4$, then also
\begin{equation}
 d_1\ge\frac{u\sqrt M}{8\sqrt2},\qquad Mu^2\le128d_1^2.
 \label{eq:returnlower}
\end{equation}
\end{theorem}
