\begin{theorem}[Excitation transport and return]\label{thm:transport}
Consider any excitation-conserving closed unitary on $n$ pure-zero inputs and a reservoir initially in $\tau^{\otimes M}$. Suppose its output marginals have trace error at most $\epsilon<1/2$ from $\tau$, and its final reservoir return error is at most $\delta>0$. Let $b=1/2-\epsilon$. Then
\begin{equation}
 M\ge
 \frac{\positive{bn/(2\delta)-n-2}^{\,2}}{2\ln(4/b)}.
 \label{eq:transport}
\end{equation}
For fixed $\epsilon<1/2$ and sufficiently small $\delta$, this is $\Omega_\epsilon(n^2/\delta^2)$ as $n$ grows. It needs only endpoint return and no output independence.
\end{theorem}
